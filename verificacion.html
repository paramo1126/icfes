const toastWrap = document.getElementById("toastWrap"); const codigoInput = document.getElementById("codigoInput"); const verificarBtn = document.getElementById("verificarBtn"); const showToast = (message, type = "error") => { const toast = document.createElement("div"); toast.className = `toast ${type}`; toast.textContent = message; toastWrap.appendChild(toast); setTimeout(() => toast.classList.add("hide"), 2800); setTimeout(() => toast.remove(), 3100); }; codigoInput.addEventListener("input", () => { codigoInput.value = codigoInput.value.replace(/\D/g, "").slice(0, 5); }); verificarBtn.addEventListener("click", () => { const savedCode = localStorage.getItem("codigo_verificacion"); const enteredCode = codigoInput.value.trim(); if (enteredCode === savedCode && /^\d{5}$/.test(enteredCode)) { window.location.href = "iniciarsesion.html"; return; } showToast("Código incorrecto, intenta de nuevo"); });
